// destination  = general_access_register || memory_address

// control
halt(argument)                          (argument doesn't do anything)
int(vector)                             (interrupt_vector)
jmp(target)                             (memory_address)

// portio

bufin(port, addr, length)               (portio_port, memory_address, length)

// arithmetic
add(src1, src2, dst)                    (operand, operand, destination)
sub(src1, src2, dst)                    (operand, operand, destination)
mul(src1, src2, dst)                    (operand, operand, destination)
div(src1, src2, dst)                    (operand, operand, destination)

// memory management
mov(dst, src)                           (destination, operand)
